// ========================================== // 1. DOM ELEMENTS // ========================================== const statusBox = document.getElementById("status"); const outputBox = document.getElementById("output"); // ========================================== // 2. HTML ESCAPE FUNCTION // ========================================== function esc(value) { return String(value ?? "").replace( /[&<>"']/g, char => ({ "&": "&", "<": "<", ">": ">", '"': """, "'": "'" }[char]) ); } // ========================================== // 3. STATUS FUNCTION // ========================================== function status(message, type = "") { statusBox.textContent = message; statusBox.className = "status" + (type ? " " + type : ""); } // ========================================== // 4. BUTTON LOADING // ========================================== function busy(value) { document.querySelectorAll("button").forEach( button => { button.disabled = value; } ); } // ========================================== // 5. RENDER TASK PLAN // ========================================== function renderPlan(plan) { if (!plan) return; let html = `

Goal

${esc(plan.goal)}

Estimated time: ${esc(plan.total_estimated_minutes)} minutes

Steps

`; (plan.steps || []).forEach(step => { html += `
${esc(step.step_number)}. ${esc(step.title)}
${esc(step.description)}
Time: ${esc(step.estimated_minutes)} min
Expected: ${esc(step.expected_output)}
`; }); // Assumptions if (plan.assumptions?.length) { html += `

Assumptions

    ${plan.assumptions.map( item => `
  • ${esc(item)}
    • ` ).join("")}
`; } // Success criteria if (plan.success_criteria?.length) { html += `

Success Criteria

    ${plan.success_criteria.map( item => `
  • ${esc(item)}
    • ` ).join("")}
`; } outputBox.innerHTML = html; } // ========================================== // 6. API REQUEST FUNCTION // ========================================== async function post(url, body) { const response = await fetch(url, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) }); const data = await response.json(); if (!response.ok) { throw new Error( data.error || "Request failed" ); } return data; } // ========================================== // 7. GENERATE TASK PLAN // ========================================== async function generatePlan() { const task = document .getElementById("task") .value .trim(); // Input validation if (task.length < 3) { status( "Enter at least 3 characters.", "error" ); return; } // Start loading busy(true); status("Generating structured task plan..."); outputBox.innerHTML = ""; try { const data = await post( "/api/plan", { request: task } ); // Render result renderPlan(data.plan); status( "Task plan generated successfully.", "success" ); } catch (error) { console.error("TaskPilot Error:", error); status( error.message, "error" ); } finally { busy(false); } }
